# ST7796S init sequence for panel-mipi-dbi
# MADCTL (0x36) is overridden by install.sh according to --rotate

command 0x11          # sleep out
delay 120
command 0xF0 0xC3     # enable command set part 1
command 0xF0 0x96     # enable command set part 2
command 0x36 0x28     # MADCTL
command 0x3A 0x55     # 16 bit/pixel
command 0xB4 0x01     # 1-dot inversion
command 0xB6 0x80 0x02 0x3B
command 0xE8 0x40 0x8A 0x00 0x00 0x29 0x19 0xA5 0x33
command 0xC1 0x06
command 0xC2 0xA7
command 0xC5 0x18
delay 120
command 0xE0 0xF0 0x09 0x0B 0x06 0x04 0x15 0x2F 0x54 0x42 0x3C 0x17 0x14 0x18 0x1B
command 0xE1 0xE0 0x09 0x0B 0x06 0x04 0x03 0x2B 0x43 0x42 0x3B 0x16 0x14 0x17 0x1B
delay 120
command 0xF0 0x3C     # disable command set part 1
command 0xF0 0x69     # disable command set part 2
delay 120
command 0x29          # display on
